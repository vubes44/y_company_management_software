<template>
  <div class="app-root">
    <h1>Przykładowe ASelect</h1>

    <section class="example">
      <ASelect
        v-model="selection1"
        :options="options"
        placeholder="Choose an option"
        name="select-default"
      />
      <p class="value">wybrano: {{ selection1 }}</p>
    </section>

    <section class="example">
      <ASelect
        v-model="selection2"
        styleType="datafield"
        :options="options"
        name="select-datafield"
      />
      <p class="value">wybrano: {{ selection2 }}</p>
    </section>

    <section class="example">
      <ASelect
        v-model="selection3"
        styleType="settings"
        :options="optionsWithDisabled"
        placeholder="Select (settings)"
        name="select-settings"
      />
      <p class="value">wybrano: {{ selection3 }}</p>
    </section>

    <section class="example">
      <ASelect :isDisabled="true" :options="options" name="select-disabled" />
      <p class="note">niedostępne</p>
    </section>

    <section class="example">
      <h2>Przykładowe MIconButton</h2>
      <div class="icon-row">
        <MIconButton iconName="search" @click="onIconClick('search')" />
        <MIconButton
          iconName="edit"
          styleType="secondary"
          @click="onIconClick('edit')"
        />
        <MIconButton
          iconName="trash"
          :isDisabled="true"
          @click="onIconClick('trash')"
        />
      </div>
      <p class="value">Search clicks: {{ clicks.search }}</p>
      <p class="value">Edit clicks: {{ clicks.edit }}</p>
      <p class="note">Trash is disabled</p>
    </section>
  </div>
</template>

<script setup lang="ts">
import { ref, reactive } from "vue";
import ASelect from "./components/atoms/ASelect.vue";
import MIconButton from "./components/molecules/MIconButton.vue";

const options = [
  { value: "one", label: "One" },
  { value: "two", label: "Two" },
  { value: "three", label: "Three" },
];

const optionsWithDisabled = [
  { value: "", label: "None", disabled: true },
  { value: "apple", label: "Apple" },
  { value: "banana", label: "Banana" },
];

const selection1 = ref<string | number | null>(null);
const selection2 = ref<string | number | null>("two");
const selection3 = ref<string | number | null>(null);

const clicks = reactive({ search: 0, edit: 0, trash: 0 });

function onIconClick(name: "search" | "edit" | "trash") {
  clicks[name]++;
}
</script>
